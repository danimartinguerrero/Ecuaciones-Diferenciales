\section{Ley de Newton del enfriamiento}

La ley de Newton describe el intercambio de calor de un cuerpo con su entorno cuando domina la convección y la diferencia de temperaturas es moderada: la temperatura del objeto varía con una velocidad proporcional a la diferencia entre su temperatura y la del ambiente.

\begin{modelo}[enfriamiento de Newton]
Sean $T(t)$ la temperatura del objeto, $T_0=T(0)$ su temperatura inicial y $S(t)$ la temperatura ambiente (\emph{source}). Entonces
\begin{equation}
  \dv{T}{t} = -k\,\bigl(T - S(t)\bigr), \qquad T(0)=T_0,
  \label{eq:newton}
\end{equation}
con $k>0$ constante de enfriamiento. Sus dimensiones son las de una frecuencia, $[k]=[\mathrm{T}]^{-1}$.
\end{modelo}

\begin{remark}
Si $T>S$ entonces $\dot T<0$ y el objeto se enfría; si $T<S$ entonces $\dot T>0$ y se calienta. En ambos casos $T$ tiende a acercarse a $S$.
\end{remark}

Reordenando términos, \eqref{eq:newton} es una ecuación lineal de primer orden con coeficiente constante:
\begin{equation}
  \dot T + k\,T = k\,S(t).
  \label{eq:newton-lineal}
\end{equation}

\begin{proposition}[solución para un ambiente arbitrario]\label{prop:newton-general}
Si $S$ es continua, la solución de \eqref{eq:newton-lineal} con $T(0)=T_0$ es
\begin{equation}
  T(t) = \ee^{-kt}\Bigl[\,T_0 + k\int_0^t \ee^{ks}\,S(s)\,\dd s\Bigr].
  \label{eq:newton-general}
\end{equation}
\end{proposition}

\begin{proof}
Multiplicando \eqref{eq:newton-lineal} por el factor integrante $\ee^{kt}$ se obtiene $\dv{}{t}\bigl(\ee^{kt}T\bigr)=k\,\ee^{kt}S(t)$. Integrando entre $0$ y $t$ y despejando $T$ se llega a \eqref{eq:newton-general}.
\end{proof}

\subsection*{Ambiente a temperatura constante}

Supongamos $S(t)\equiv S_0$. La ecuación \eqref{eq:newton-lineal} se resuelve en dos etapas.

\begin{algoritmo}[ecuación lineal $\dot y + a\,y = b(t)$]
\begin{pasos}
  \item \emph{Homogénea.} $\dot T_h + kT_h=0$ es de variables separables: $\dd T_h/T_h=-k\,\dd t$, luego $T_h = C\,\ee^{-kt}$.
  \item \emph{Variación de la constante (Lagrange).} Se busca $T = C(t)\,\ee^{-kt}$. Al sustituir, los términos en $C$ se cancelan y queda $\dot C\,\ee^{-kt}=kS_0$, es decir, $C(t)=S_0\,\ee^{kt}+c_0$.
  \item \emph{Solución general.} $T(t)=S_0 + c_0\,\ee^{-kt}$. Alternativamente, $T_p\equiv S_0$ se halla por inspección (solución estacionaria, $\dot T=0$) y $T=T_h+T_p$.
  \item \emph{Condición inicial.} $T(0)=T_0$ da $c_0=T_0-S_0$.
\end{pasos}
\end{algoritmo}

\begin{proposition}[solución con ambiente constante]\label{prop:newton}
La solución del problema \eqref{eq:newton} con $S\equiv S_0$ es
\begin{equation}
  T(t) = S_0 + (T_0 - S_0)\,\ee^{-kt},
  \label{eq:newton-sol}
\end{equation}
y $T(t)\to S_0$ cuando $t\to\infty$.
\end{proposition}

\begin{proof}
Es el resultado del algoritmo anterior. También se obtiene de \eqref{eq:newton-general}, ya que $k\int_0^t \ee^{ks}S_0\,\dd s=S_0(\ee^{kt}-1)$. Como $k>0$, el término exponencial tiende a cero.
\end{proof}

\begin{remark}
La magnitud $\tau=1/k$ es la \emph{constante de tiempo}: la diferencia $T-S_0$ se reduce en un factor $\ee^{-1}$ cada intervalo $\tau$. La convergencia a $S_0$ es asintótica: $T(t)=S_0$ solo en el límite $t\to\infty$ (o, a efectos prácticos, para $t$ suficientemente grande).
\end{remark}

\begin{example}
Un cuerpo a $T_0=\SI{90}{\celsius}$ se sitúa en un entorno a $S_0=\SI{20}{\celsius}$. Tras \SI{10}{\minute} su temperatura es \SI{60}{\celsius}. Determinar $k$ y el instante en que alcanza \SI{30}{\celsius}.

De \eqref{eq:newton-sol}, $60 = 20 + 70\,\ee^{-10k}$, luego
\[
  k = \tfrac{1}{10}\ln\tfrac{70}{40} \approx \SI{0,056}{\per\minute}.
\]
Imponiendo $T(t)=30$: $\ee^{-kt}=\tfrac{10}{70}$, de donde
$t = \tfrac{1}{k}\ln 7 \approx \SI{34,8}{\minute}$.\finejemplo
\end{example}

Cuando el ambiente varía con el tiempo, \eqref{eq:newton-general} sigue siendo válida. Los casos $S(t)=S_0+b\sin\omega t$ y $S(t)=S_0\,\ee^{-\omega t}$ se resuelven con detalle, por varios métodos, en la \cref{sec:seminarios} (problemas~1 y~2).
